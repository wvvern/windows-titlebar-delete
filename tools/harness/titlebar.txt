# Titlebar test for tools/harness.ps1, which runs on the title screen with the default config.
# The core harness skips lines that start with "#".
# A "# expect <bit> <on|off> ..." line tells harness.ps1 what the next "style" line must show.
# The bits are caption (0x00C00000) and thickframe (0x00040000).
# After each "style", "wait 40" keeps the window stable while harness.ps1 reads the style from outside the game.
wait 100
# expect caption off thickframe on
style
wait 40
toggle
# expect caption on thickframe on
style
wait 40
toggle
# expect caption off
style
wait 40
fullscreen on
wait 100
fullscreen off
wait 100
# expect caption off
style
wait 40
set keepResizeBorder off
# expect caption off thickframe off
style
wait 40
set keepResizeBorder on
settings
wait 40
shot settings
wait 10
click Hide the titlebar at startup
wait 10
shot settings-off
wait 10
click Hide the titlebar at startup
wait 10
click Done
wait 20
# expect caption off thickframe on
style
wait 40
quit
